<body>

  <div id="game-container">
    <div id="timer">0.0초</div>
    <div id="dino"></div>
    <div id="cactus"></div>
    <button id="start-btn" onclick="startGame()">게임 시작</button>
  </div>

  <script>
    const dino = document.getElementById("dino");
    const cactus = document.getElementById("cactus");
    const timerElement = document.getElementById("timer");
    const startBtn = document.getElementById("start-btn");

    let isJumping = false;
    let isGaming = false;
    let time = 0;
    let timerInterval = null;
    let checkCollisionInterval = null;

    // 스페이스바 점프 이벤트
    document.addEventListener("keydown", function(event) {
      if (event.code === "Space" && !isJumping && isGaming) {
        jump();
      }
    });

    function jump() {
      isJumping = true;
      dino.classList.add("jump");
      setTimeout(() => {
        dino.classList.remove("jump");
        isJumping = false;
      }, 500);
    }

    // 게임 시작 함수
    function startGame() {
      // 초기화
      isGaming = true;
      time = 0;
      timerElement.innerText = "0.0초";
      startBtn.style.display = "none";
      cactus.classList.add("move-cactus");

      // 0.1초 타이머 작동
      timerInterval = setInterval(() => {
        time += 0.1;
        timerElement.innerText = time.toFixed(1) + "초";
      }, 100);

      // 충돌 감지 (게임 오버 판정)
      checkCollisionInterval = setInterval(() => {
        const dinoBottom = parseInt(window.getComputedStyle(dino).getPropertyValue("bottom"));
        const cactusLeft = parseInt(window.getComputedStyle(cactus).getPropertyValue("left"));

        // 공룡과 선인장이 겹쳤을 때
        if (cactusLeft > 50 && cactusLeft < 90 && dinoBottom <= 70) {
          endGame();
        }
      }, 10);
    }

    // 게임 종료 처리
    function endGame() {
      isGaming = false;
      cactus.classList.remove("move-cactus");
      clearInterval(timerInterval);
      clearInterval(checkCollisionInterval);
      alert(`게임 오버! 기록: ${time.toFixed(1)}초`);
      startBtn.innerText = "다시 시작";
      startBtn.style.display = "block";
    }
  </script>
