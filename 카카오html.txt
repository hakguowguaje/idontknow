  <!-- 카카오맵 JavaScript API 로드 (autoload=false를 사용하여 지도가 준비된 후 실행되도록 설정) -->

  <!-- 메인 화면 구조 (왼쪽: 사이드바, 오른쪽: 지도) -->
  <div class="container">
    <!-- 왼쪽 사이드바 -->
    <div class="sidebar">
      <div class="search-box">
        <input type="text" id="keyword" placeholder="장소를 입력하세요..." onkeydown="if(event.key === 'Enter') searchPlaces()">
        <button onclick="searchPlaces()">검색</button>

  <!-- JavaScript 소스코드 -->
  <script>
    // 전역 변수 선언
    let map;
    let placesService;
    let markers = [];
    let infoWindow;

    // 카카오 SDK 로드가 완료된 후 안전하게 실행하기 위해 kakao.maps.load() 사용
    kakao.maps.load(function() {
      initMap();
    });

    // 지도 초기화 함수
    function initMap() {
      const container = document.getElementById('map');
      
      // 기본 위치 (위치 권한을 거부하거나 오류 발생 시 사용할 기본 좌표: 서울시청)
      let defaultLat = 37.566826;
      let defaultLng = 126.9786567;

      const options = {
        center: new kakao.maps.LatLng(defaultLat, defaultLng),
        level: 5 // 요구사항: 지도 확대 레벨 5
      };

      // 지도 생성
      map = new kakao.maps.Map(container, options);
      
      // 장소 검색 객체 생성 및 인포윈도우 생성
      placesService = new kakao.maps.services.Places();
      infoWindow = new kakao.maps.InfoWindow({ zIndex: 1 });

      // HTML5 Geolocation을 이용해 사용자 위치 받아오기
      if (navigator.geolocation) {
        navigator.geolocation.getCurrentPosition(
          function(position) {
            const currentLat = position.coords.latitude;
            const currentLng = position.coords.longitude;
            const currentPosition = new kakao.maps.LatLng(currentLat, currentLng);

            // 지도 중심을 현재 위치로 이동
            map.setCenter(currentPosition);

            // 현재 위치 마커 표시
            const currentMarker = new kakao.maps.Marker({
              map: map,
              position: currentPosition
            });

            // 현재 위치에 간단한 메시지 표시
            const locationContent = '<div style="padding:5px;font-size:12px;">현재 위치</div>';
            infoWindow.setContent(locationContent);
            infoWindow.open(map, currentMarker);
          },
          function(error) {
            console.warn('위치 정보를 가져오는 데 실패했습니다. 기본 위치로 설정합니다.', error);
          }
        );
      } else {
        alert('이 브라우저는 위치 서비스를 지원하지 않습니다.');
      }
    }

    // 장소 검색 함수
    function searchPlaces() {
      const keyword = document.getElementById('keyword').value;

      if (!keyword.replace(/^\s+|\s+$/g, '')) {
        alert('키워드를 입력해 주세요!');
        return;
      }

      // 카카오 키워드 검색 API 호출
      placesService.keywordSearch(keyword, placesSearchCB);
    }

    // 키워드 검색 완료 시 호출되는 콜백 함수
    function placesSearchCB(data, status, pagination) {
      if (status === kakao.maps.services.Status.OK) {
        // 검색 결과를 사이드바에 표시하고 지도 마커 업데이트
        displayPlaces(data);
      } else if (status === kakao.maps.services.Status.ZERO_RESULT) {
        alert('검색 결과가 존재하지 않습니다.');
      } else if (status === kakao.maps.services.Status.ERROR) {
        alert('검색 결과 중 오류가 발생했습니다.');
      }
    }

    // 검색 결과 목록과 마커를 화면에 표시하는 함수
    function displayPlaces(places) {
      const listElement = document.getElementById('placesList');
      const bounds = new kakao.maps.LatLngBounds();

      // 기존 리스트 및 마커 초기화
      removeAllMarkers();
      listElement.innerHTML = '';

      for (let i = 0; i < places.length; i++) {
        const placePosition = new kakao.maps.LatLng(places[i].y, places[i].x);
        const marker = addMarker(placePosition);
        const itemElement = createListItem(i, places[i]);

        // 지도 검색 범위에 해당 위치 추가
        bounds.extend(placePosition);

        // 검색 결과 항목 또는 마커를 클릭했을 때 인포윈도우(정보창) 표시
        (function(marker, place) {
          kakao.maps.event.addListener(marker, 'click', function() {
            displayInfoWindow(marker, place.place_name);
          });

          itemElement.onclick = function() {
            displayInfoWindow(marker, place.place_name);
            map.panTo(new kakao.maps.LatLng(place.y, place.x));
          };
        })(marker, places[i]);

        listElement.appendChild(itemElement);
      }

      // 검색된 장소들이 한눈에 보이도록 지도 영역 조정
      map.setBounds(bounds);
    }

    // 검색 결과 리스트 아이템 HTML 생성 함수
    function createListItem(index, place) {
      const el = document.createElement('li');
      el.className = 'result-item';
      
      let itemStr = `<h4>${place.place_name}</h4>`;
      if (place.road_address_name) {
        itemStr += `<p>${place.road_address_name}</p>`;
      } else {
        itemStr += `<p>${place.address_name}</p>`;
      }
      
      el.innerHTML = itemStr;
      return el;
    }

    // 마커 생성 함수
    function addMarker(position) {
      const marker = new kakao.maps.Marker({
        position: position
      });
      marker.setMap(map);
      markers.push(marker);
      return marker;
    }

    // 기존 마커 모두 제거 함수
    function removeAllMarkers() {
      for (let i = 0; i < markers.length; i++) {
        markers[i].setMap(null);
      }
      markers = [];
    }

    // 마커 클릭 시 정보창(InfoWindow) 띄우는 함수
    function displayInfoWindow(marker, title) {
      const content = `<div style="padding:5px;font-size:12px;z-index:1;">${title}</div>`;
      infoWindow.setContent(content);
      infoWindow.open(map, marker);
    }
  </script>
